cmake_minimum_required(VERSION 3.16)
project(NeoLspPlanner LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml Test)
find_package(Threads REQUIRED)
get_filename_component(NEO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
qt_add_resources(LSP_RESOURCES ${NEO_ROOT}/third-party/lsp-planner/neo_lsp_planner.qrc)
add_executable(TestLspPlanner ../testlspplanner.cpp ../testlspplanner.h
    ${NEO_ROOT}/qt-models/lspplanner.cpp ${NEO_ROOT}/qt-models/lspplanner.h ${LSP_RESOURCES})
target_include_directories(TestLspPlanner PRIVATE ${NEO_ROOT})
target_compile_definitions(TestLspPlanner PRIVATE SUBSURFACE_TEST_DATA="${NEO_ROOT}")
target_link_libraries(TestLspPlanner PRIVATE Qt6::Core Qt6::Qml Qt6::Test Threads::Threads)
enable_testing()
add_test(NAME TestLspPlanner COMMAND TestLspPlanner)
